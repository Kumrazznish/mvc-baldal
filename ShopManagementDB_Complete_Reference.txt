---------------------------------------------------------------------------
TABLE 1 : Stores
---------------------------------------------------------------------------
Purpose : Shop/branch ki basic information store karta hai.
          Ek store create hota hai — Admin usi ke under kaam karta hai.

CREATE TABLE dbo.Stores
(
    StoreId     INT           IDENTITY(1,1) NOT NULL,   -- Auto increment PK
    StoreName   NVARCHAR(150) NOT NULL,                 -- Store ka naam (e.g. "Main Branch")
    StoreCode   NVARCHAR(50)  NOT NULL,                 -- Unique short code (e.g. "ST001")
    Address     NVARCHAR(500) NULL,                     -- Pata (optional)
    Phone       NVARCHAR(20)  NULL,                     -- Phone number (optional)
    Email       NVARCHAR(150) NULL,                     -- Store ka email (optional)
    IsActive    BIT           NOT NULL DEFAULT (1),     -- 1=Active, 0=Inactive
    CreatedDate DATETIME      NOT NULL DEFAULT (GETDATE()), -- Kab create hua
    UpdatedDate DATETIME      NULL,                     -- Last update kab hua

    CONSTRAINT PK_Stores          PRIMARY KEY (StoreId),
    CONSTRAINT UQ_Stores_StoreCode UNIQUE (StoreCode)  -- StoreCode duplicate nahi ho sakta
);

Column Explanation:
    StoreId    : Database khud auto number generate karta hai (1, 2, 3...)
    StoreCode  : Unique hai — same code 2 stores mein nahi ho sakta
    IsActive   : Default 1 (active). Deactivate karo toh 0
    UpdatedDate: NULL start mein, edit karne par GETDATE() set hota hai

TABLE 2 : Users
---------------------------------------------------------------------------
Purpose : Sab login users yahan stored hain — Admin aur Employee dono.
          Email se login hota hai (pehle Username tha, ab Email hai).

CREATE TABLE dbo.Users
(
    UserId       INT           IDENTITY(1,1) NOT NULL,  -- Auto increment PK
    Name         NVARCHAR(100) NOT NULL,                -- Full naam (e.g. "Rahul Sharma")
    Email        NVARCHAR(150) NOT NULL,                -- Login ke liye email (UNIQUE)
    PasswordHash NVARCHAR(500) NOT NULL,                -- Password
    Role         NVARCHAR(20)  NOT NULL,                -- "Admin" ya "Employee"
    StoreId      INT           NULL,                    -- Employee kis store mein hai (Admin = NULL)
    IsActive     BIT           NOT NULL DEFAULT (1),    -- 1=Active, 0=Blocked
    CreatedDate  DATETIME      NOT NULL DEFAULT (GETDATE()),
    UpdatedDate  DATETIME      NULL,

    CONSTRAINT PK_Users         PRIMARY KEY (UserId),
    CONSTRAINT UQ_Users_Email   UNIQUE (Email),
    CONSTRAINT CK_Users_Role    CHECK (Role IN ('Admin', 'Employee')),
    CONSTRAINT FK_Users_Stores  FOREIGN KEY (StoreId) REFERENCES dbo.Stores(StoreId)
);

Column Explanation:
    Email   : Login field. Har user ka alag email hona chahiye.
    Role    : "Admin" = Full access. "Employee" = Invoice banana only.
    StoreId : Admin ke liye NULL. Employee ke liye mandatory.
    IsActive: 0 karo toh employee login nahi kar sakta.

Constraints:
    UQ_Users_Email  : Same email 2 baar insert nahi ho sakta — error aayega
    CK_Users_Role   : Sirf "Admin" ya "Employee" allowed — kuch aur dalo toh error
    FK_Users_Stores : StoreId woh StoreId hona chahiye jo Stores table mein exist kare

TABLE 3 : Products
---------------------------------------------------------------------------
Purpose : Shop ke sab products. Quantity real-time update hoti hai invoice par.

CREATE TABLE dbo.Products
(
    ProductId   INT            IDENTITY(1,1) NOT NULL,
    ProductCode NVARCHAR(50)   NOT NULL,                -- Unique code (e.g. "P001")
    ProductName NVARCHAR(200)  NOT NULL,                -- Product naam
    Quantity    INT            NOT NULL,                -- Current stock
    Price       DECIMAL(18,2)  NOT NULL,                -- Price (e.g. 10.00)
    ImagePath   NVARCHAR(500)  NULL,                    -- /Uploads/Products/guid.jpg
    IsActive    BIT            NOT NULL DEFAULT (1),    -- Soft delete flag
    CreatedDate DATETIME       NOT NULL DEFAULT (GETDATE()),
    UpdatedDate DATETIME       NULL,

    CONSTRAINT PK_Products             PRIMARY KEY (ProductId),
    CONSTRAINT UQ_Products_ProductCode UNIQUE (ProductCode),
    CONSTRAINT CK_Products_Quantity    CHECK (Quantity >= 0),
    CONSTRAINT CK_Products_Price       CHECK (Price    >= 0)
);

Column Explanation:
    ProductCode: "P001", "P002" etc. Import mein agar exist karta hai toh update hoga
    Quantity   : Invoice create hone par SP automatically ghatata hai
    Price      : Browser se price trust nahi hota — SP DB se read karta hai
    IsActive   : Delete par 0 hota hai (soft delete). Data physically nahi hata.

Indexes:
    IX_Products_ProductName  -- Search fast
    IX_Products_IsActive     -- Filter fast

SP 1 : SP_Account_Login
Purpose : Email + Password se login validate karna
Called by: AccountDAL.GetUserByEmail()
===========================================================================

CREATE OR ALTER PROCEDURE dbo.SP_Account_Login
    @Email    NVARCHAR(150),
    @Password NVARCHAR(255)
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        UserId,       -- Session mein store hoga
        Name,         -- Header mein dikhayi dega
        Email,        -- Session mein store hoga
        Role,         -- "Admin" ya "Employee" — redirect decide karta hai
        StoreId,      -- Employee ka store (Admin ke liye NULL)
        IsActive
    FROM dbo.Users
    WHERE Email        = @Email
      AND PasswordHash = @Password
      AND IsActive     = 1;  -- Sirf active users login kar sakte hain
END;

Flow:
    1. User form mein email + password dalta hai
    2. AccountController → AccountBAL.ValidateLogin()
    3. AccountDAL.GetUserByEmail() → SP call
    4. Agar row aaya → Session set karo (UserId, Name, Email, Role, StoreId)
    5. Role="Admin" → Dashboard/Index
       Role="Employee" → Employee/Index
    6. Koi row nahi → "Invalid email or password"

SP 4 : SP_Product_Save
Purpose : Product add/restock. Exist kare toh UPDATE, warna INSERT (UPSERT logic)
Called by: ProductDAL.SaveProduct() — Add Product form + Excel Import
===========================================================================

CREATE OR ALTER PROCEDURE dbo.SP_Product_Save
    @ProductCode NVARCHAR(50),
    @ProductName NVARCHAR(200),
    @Quantity    INT,
    @Price       DECIMAL(18,2),
    @ImagePath   NVARCHAR(500) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS (
        SELECT 1 FROM dbo.Products WHERE ProductCode = @ProductCode
    )
    BEGIN
        -- ProductCode exist karta hai = UPDATE
        UPDATE dbo.Products
        SET
            ProductName = @ProductName,
            Quantity    = Quantity + @Quantity,   -- ADD karo (replace nahi)
            Price       = @Price,
            ImagePath   = CASE
                              WHEN @ImagePath IS NULL OR @ImagePath = ''
                              THEN ImagePath      -- Purana image rakho
                              ELSE @ImagePath     -- Naya image set karo
                          END,
            IsActive    = 1,                      -- Agar delete tha toh reactivate
            UpdatedDate = GETDATE()
        WHERE ProductCode = @ProductCode;
    END
    ELSE
    BEGIN
        -- Naya product = INSERT
        INSERT INTO dbo.Products
        (ProductCode, ProductName, Quantity, Price, ImagePath, IsActive, CreatedDate)
        VALUES
        (@ProductCode, @ProductName, @Quantity, @Price, @ImagePath, 1, GETDATE());
    END
END;

Important:
    - Quantity ADD hoti hai REPLACE nahi (stock management ke liye)
    - Image NULL = purana image safe rahega
    - Deleted product same code se add karo = reactivate hoga

SP 11 : SP_Store_Update
Purpose : Store edit karna
Called by: StoreDAL.UpdateStore()
===========================================================================

CREATE OR ALTER PROCEDURE dbo.SP_Store_Update
(
    @StoreId   INT,
    @StoreName NVARCHAR(150),
    @StoreCode NVARCHAR(50),
    @Address   NVARCHAR(500) = NULL,
    @Phone     NVARCHAR(20)  = NULL,
    @Email     NVARCHAR(150) = NULL,
    @IsActive  BIT
)
AS
BEGIN
    SET NOCOUNT ON;

    IF NOT EXISTS (SELECT 1 FROM dbo.Stores WHERE StoreId = @StoreId)
        THROW 50002, 'Store not found.', 1;

    IF EXISTS (
        SELECT 1 FROM dbo.Stores
        WHERE StoreCode = @StoreCode AND StoreId <> @StoreId
    )
        THROW 50003, 'Store code already exists.', 1;

    UPDATE dbo.Stores
    SET StoreName=@StoreName, StoreCode=@StoreCode, Address=@Address,
        Phone=@Phone, Email=@Email, IsActive=@IsActive, UpdatedDate=GETDATE()
    WHERE StoreId = @StoreId;

    SELECT @StoreId AS StoreId;
END;

===========================================================================
SP 15 : SP_Invoice_CreateComplete   *** SABSE IMPORTANT SP ***
Purpose : Pura invoice ek atomic transaction mein. 14 steps.
Called by: InvoiceDAL.CreateCompleteInvoice()
Returns : InvoiceId, InvoiceNo, TotalAmount
===========================================================================

CREATE OR ALTER PROCEDURE dbo.SP_Invoice_CreateComplete
(
    @CreatedBy     INT,
    @CustomerName  NVARCHAR(150),
    @CustomerPhone NVARCHAR(20) = NULL,
    @InvoiceItems  dbo.InvoiceItemType READONLY  -- TVP (Table-Valued Parameter)
)
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;  -- Koi bhi error = automatic rollback

    DECLARE @InvoiceId   INT;
    DECLARE @InvoiceNo   NVARCHAR(30);
    DECLARE @TotalAmount DECIMAL(18,2);

    BEGIN TRY
        BEGIN TRANSACTION;

        -- STEP 1: User validate karo
        IF @CreatedBy <= 0
            THROW 50001, 'Invalid user.', 1;
        IF NOT EXISTS (SELECT 1 FROM dbo.Users WHERE UserId = @CreatedBy AND IsActive = 1)
            THROW 50002, 'Invalid or inactive user.', 1;

        -- STEP 2: Customer naam validate karo
        SET @CustomerName = LTRIM(RTRIM(@CustomerName));
        IF NULLIF(@CustomerName, '') IS NULL
            THROW 50003, 'Customer name is required.', 1;

        -- STEP 3: Koi item select kiya?
        IF NOT EXISTS (SELECT 1 FROM @InvoiceItems)
            THROW 50004, 'Please select at least one product.', 1;

        -- STEP 4: Duplicate product check
        IF EXISTS (
            SELECT ProductId FROM @InvoiceItems GROUP BY ProductId HAVING COUNT(*) > 1
        )
            THROW 50005, 'Same product cannot be added multiple times.', 1;

        -- STEP 5: Stock check
        IF EXISTS (
            SELECT 1 FROM @InvoiceItems I
            LEFT JOIN dbo.Products P ON P.ProductId = I.ProductId
            WHERE P.ProductId IS NULL OR P.IsActive = 0
               OR I.ProductId <= 0 OR I.Quantity <= 0
               OR P.Quantity < I.Quantity
        )
            THROW 50006, 'One or more products invalid or insufficient stock.', 1;

        -- STEP 6: Total CALCULATE KARO — DATABASE SE PRICE (browser pe trust nahi)
        SELECT @TotalAmount = SUM(CAST(P.Price * I.Quantity AS DECIMAL(18,2)))
        FROM @InvoiceItems I
        INNER JOIN dbo.Products P ON P.ProductId = I.ProductId;

        IF @TotalAmount IS NULL OR @TotalAmount <= 0
            THROW 50007, 'Invoice total must be greater than zero.', 1;

        -- STEP 7: Invoice header insert (temporary InvoiceNo)
        INSERT INTO dbo.Invoices
        (InvoiceNo, CustomerName, CustomerPhone, CreatedBy, InvoiceDate, TotalAmount)
        VALUES
        ('TEMP', @CustomerName, NULLIF(LTRIM(RTRIM(@CustomerPhone)),''), @CreatedBy, GETDATE(), @TotalAmount);

        -- STEP 8: New ID capture
        SET @InvoiceId = CONVERT(INT, SCOPE_IDENTITY());

        -- STEP 9: InvoiceNo generate karo  Format: INV-20261005-000001
        SET @InvoiceNo =
            'INV-' + CONVERT(CHAR(8), GETDATE(), 112) + '-' +
            RIGHT('000000' + CAST(@InvoiceId AS NVARCHAR(6)), 6);

        -- STEP 10: InvoiceNo update karo
        UPDATE dbo.Invoices SET InvoiceNo = @InvoiceNo WHERE InvoiceId = @InvoiceId;

        -- STEP 11: Line items insert (DB se price lo)
        INSERT INTO dbo.InvoiceDetails
        (InvoiceId, ProductId, ProductName, Quantity, UnitPrice, TotalAmount)
        SELECT @InvoiceId, I.ProductId, P.ProductName, I.Quantity,
               P.Price, CAST(P.Price * I.Quantity AS DECIMAL(18,2))
        FROM @InvoiceItems I
        INNER JOIN dbo.Products P ON P.ProductId = I.ProductId;

        -- STEP 12: Stock deduct karo
        UPDATE P
        SET P.Quantity = P.Quantity - I.Quantity, P.UpdatedDate = GETDATE()
        FROM dbo.Products P
        INNER JOIN @InvoiceItems I ON I.ProductId = P.ProductId;

        -- STEP 13: COMMIT
        COMMIT TRANSACTION;

        -- STEP 14: Result return karo
        SELECT @InvoiceId AS InvoiceId, @InvoiceNo AS InvoiceNo, @TotalAmount AS TotalAmount;

    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        THROW;  -- Error C# ko forward karo
    END CATCH
END;

Error Codes:
    50001 = Invalid user ID
    50002 = User inactive ya exist nahi
    50003 = Customer name missing
    50004 = Koi product select nahi kiya
    50005 = Duplicate product in list
    50006 = Invalid product / insufficient stock
    50007 = Total amount zero ya negative

CREATE OR ALTER PROCEDURE dbo.SP_Invoice_GetById
(
    @InvoiceId INT
)
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        I.InvoiceId, I.InvoiceNo, I.CustomerName, I.CustomerPhone,
        I.CreatedBy,
        U.Name AS CreatedByName,   -- Employee naam JOIN se
        I.InvoiceDate, I.TotalAmount

END;
